// API ID const STOP_ID = "001552"; const ITEMS_PER_PAGE = 12; let routesData = []; let currentPage = 0; let autoPageTimer = null; // clock function updateClock() { const now = new Date(); const hours = String(now.getHours()).padStart(2, '0'); const minutes = String(now.getMinutes()).padStart(2, '0'); const seconds = String(now.getSeconds()).padStart(2, '0'); document.getElementById('currentTime').textContent = `${hours}:${minutes}:${seconds}`; } // formatETA(etaTimeString) function formatETA(etaTimeString) { if (!etaTimeString) return "-"; const etaDate = new Date(etaTimeString); if (isNaN(etaDate.getTime())) return "-"; const h = String(etaDate.getHours()).padStart(2, '0'); const m = String(etaDate.getMinutes()).padStart(2, '0'); return `${h}:${m}`; } // Get ETA from API async function fetchETADefault() { try { const response = await fetch(`https://data.etabus.gov.hk/v1/transport/kmb/stop-eta/${STOP_ID}`); const result = await response.json(); if (!result.data || !Array.isArray(result.data)) return; // get data const map = new Map(); result.data.forEach(item => { // items if (!map.has(item.route) && item.eta) { map.set(item.route, { route: item.route, destZh: `往 ${item.dest_tc}`, destEn: `TO ${item.dest_en.toUpperCase()}`, eta: formatETA(item.eta), rawEta: item.eta }); } }); // A30, A31, A32... NA31 routesData = Array.from(map.values()).sort((a, b) => a.route.localeCompare(b.route, undefined, { numeric: true, sensitivity: 'base' }) ); // render renderPage(currentPage); } catch (error) { console.error("無法取得即時巴士資料：", error); } } // renderPage(pageIndex) function renderPage(pageIndex) { const totalPages = Math.ceil(routesData.length / ITEMS_PER_PAGE) || 1; currentPage = pageIndex % totalPages; if (currentPage < 0) currentPage = totalPages - 1; const grid = document.getElementById('routeGrid'); grid.innerHTML = ''; const startIndex = currentPage * ITEMS_PER_PAGE; const pageItems = routesData.slice(startIndex, startIndex + ITEMS_PER_PAGE); for (let i = 0; i < ITEMS_PER_PAGE; i++) { const item = pageItems[i]; const card = document.createElement('div'); card.className = 'route-card'; if (item) { card.innerHTML = `
預計時間 ETA
${item.route}
${item.destZh}
${item.destEn}
${item.eta}
`; } else { card.innerHTML = `
-
`; } grid.appendChild(card); } document.getElementById('pageIndicator').textContent = `第 ${currentPage + 1} 頁 / 共 ${totalPages} 頁 (每 10 秒自動翻頁)`; } function nextPage() { renderPage(currentPage + 1); resetAutoPlay(); } function prevPage() { renderPage(currentPage - 1); resetAutoPlay(); } function resetAutoPlay() { clearInterval(autoPageTimer); autoPageTimer = setInterval(() => { renderPage(currentPage + 1); }, 10000); // 10 秒切換下一頁 } // fresh setInterval(updateClock, 1000); updateClock(); // 30s API fetchETADefault(); setInterval(fetchETADefault, 30000); resetAutoPlay();
